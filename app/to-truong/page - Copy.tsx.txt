'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/utils/supabase/client'
import { Award, LogOut, Upload, CheckCircle2, Image as ImageIcon } from 'lucide-react'

export default function ToTruongDashboard() {
  const [currentUser, setCurrentUser] = useState<any>(null)
  const [targetTeam, setTargetTeam] = useState<number>(0)
  const [students, setStudents] = useState<any[]>([])
  const [weeks, setWeeks] = useState<any[]>([])
  const [selectedWeek, setSelectedWeek] = useState<string>('')
  const [scores, setScores] = useState<{ [key: string]: { score: number; eval: string; image_url: string } }>({})
  const [loading, setLoading] = useState(false)

  const router = useRouter()
  const supabase = createClient()

  useEffect(() => {
    const userStr = localStorage.getItem('currentUser')
    if (!userStr) {
      router.push('/login')
      return
    }
    const user = JSON.parse(userStr)
    if (!user.role.startsWith('to_truong_')) {
      router.push('/login')
      return
    }
    setCurrentUser(user)

    // Xác định tổ cần chấm chéo theo quy tắc
    // Tổ 1 -> Chấm Tổ 2, Tổ 2 -> Chấm Tổ 1, Tổ 3 -> Chấm Tổ 4, Tổ 4 -> Chấm Tổ 3
    const tNum = user.team_number
    let mappedTarget = 1
    if (tNum === 1) mappedTarget = 2
    else if (tNum === 2) mappedTarget = 1
    else if (tNum === 3) mappedTarget = 4
    else if (tNum === 4) mappedTarget = 3

    setTargetTeam(mappedTarget)
    fetchInitialData(mappedTarget)
  }, [])

  const fetchInitialData = async (targetT: number) => {
    setLoading(true)
    // Lấy danh sách tuần
    const { data: weeksData } = await supabase.from('weeks').select('*')
    if (weeksData && weeksData.length > 0) {
      setWeeks(weeksData)
      setSelectedWeek(weeksData[0].id)
      fetchScoresAndStudents(targetT, weeksData[0].id)
    }
    setLoading(false)
  }

  const fetchScoresAndStudents = async (targetT: number, weekId: string) => {
    // Lấy học sinh thuộc tổ mục tiêu
    const { data: studentsData } = await supabase
      .from('students')
      .select('*')
      .eq('team_number', targetT)
      .order('full_name', { ascending: true })

    if (studentsData) setStudents(studentsData)

    // Lấy điểm thi đua đã chấm của tuần đó
    const { data: scoresData } = await supabase
      .from('emulation_scores')
      .select('*')
      .eq('week_id', weekId)

    if (scoresData) {
      const scoreMap: any = {}
      scoresData.forEach((s: any) => {
        scoreMap[s.student_id] = {
          score: s.score,
          eval: s.evaluation,
          image_url: s.image_url || ''
        }
      })
      setScores(scoreMap)
    }
  }

  const handleWeekChange = (weekId: string) => {
    setSelectedWeek(weekId)
    fetchScoresAndStudents(targetTeam, weekId)
  }

  const calculateEvaluation = (score: number) => {
    if (score < 50) return 'Y'
    if (score <= 64) return 'TB'
    if (score <= 79) return 'K'
    return 'T'
  }

  const handleScoreChange = (studentId: string, val: string) => {
    const num = parseFloat(val) || 0
    const ev = calculateEvaluation(num)
    setScores(prev => ({
      ...prev,
      [studentId]: {
        ...prev[studentId],
        score: num,
        eval: ev
      }
    }))
  }

  const handleImageUpload = async (studentId: string, e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    setLoading(true)
    const fileName = `vi_pham_${Date.now()}_${file.name}`
    const { data, error } = await supabase.storage.from('thi-dua').upload(fileName, file)

    if (error) {
      alert('Lỗi tải ảnh lên!')
      setLoading(false)
      return
    }

    const { data: publicUrlData } = supabase.storage.from('thi-dua').getPublicUrl(fileName)
    const publicUrl = publicUrlData.publicUrl

    setScores(prev => ({
      ...prev,
      [studentId]: {
        ...prev[studentId],
        image_url: publicUrl
      }
    }))
    setLoading(false)
    alert('Tải ảnh vi phạm thành công!')
  }

  const handleSaveAll = async () => {
    if (!selectedWeek) return
    setLoading(true)

    for (const student of students) {
      const studentScore = scores[student.id] || { score: 100, eval: 'T', image_url: '' }
      
      // Kiểm tra xem đã có bản ghi chưa
      const { data: existing } = await supabase
        .from('emulation_scores')
        .select('id')
        .eq('student_id', student.id)
        .eq('week_id', selectedWeek)
        .single()

      if (existing) {
        await supabase
          .from('emulation_scores')
          .update({
            score: studentScore.score,
            evaluation: studentScore.eval,
            image_url: studentScore.image_url,
            updated_by: currentUser.username
          })
          .eq('id', existing.id)
      } else {
        await supabase.from('emulation_scores').insert([
          {
            student_id: student.id,
            week_id: selectedWeek,
            score: studentScore.score,
            evaluation: studentScore.eval,
            image_url: studentScore.image_url,
            updated_by: currentUser.username
          }
        ])
      }
    }

    setLoading(false)
    alert('Đã lưu điểm thi đua thành công!')
  }

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      {/* Header */}
      <header className="bg-white border-b border-purple-100 shadow-xs px-6 py-4 flex justify-between items-center">
        <div className="flex items-center space-x-3">
          <div className="bg-purple-600 text-white p-2.5 rounded-2xl shadow-md">
            <Award className="h-6 w-6" />
          </div>
          <div>
            <h1 className="text-lg font-black text-gray-900">Khu vực Chấm điểm của Tổ Trưởng</h1>
            <p className="text-xs text-purple-600 font-bold">
              {currentUser?.full_name} — Đang chấm cho <span className="text-red-600 uppercase font-extrabold">Tổ {targetTeam}</span> (Chấm chéo)
            </p>
          </div>
        </div>
        <button
          onClick={() => { localStorage.removeItem('currentUser'); router.push('/login'); }}
          className="flex items-center space-x-2 bg-red-50 hover:bg-red-100 text-red-600 px-4 py-2 rounded-xl text-xs font-bold transition cursor-pointer"
        >
          <LogOut className="h-4 w-4" />
          <span>Đăng xuất</span>
        </button>
      </header>

      {/* Main Container */}
      <main className="flex-1 max-w-6xl w-full mx-auto p-6 space-y-6">
        <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 flex flex-col md:flex-row justify-between items-center gap-4">
          <div className="flex items-center space-x-3 w-full md:w-auto">
            <span className="text-xs font-extrabold text-gray-700">Chọn Tuần:</span>
            <select
              value={selectedWeek}
              onChange={(e) => handleWeekChange(e.target.value)}
              className="px-4 py-2.5 bg-purple-50 border border-purple-200 rounded-xl font-bold text-purple-900 text-xs cursor-pointer outline-none"
            >
              {weeks.map((w) => (
                <option key={w.id} value={w.id}>{w.week_name} ({w.period})</option>
              ))}
            </select>
          </div>

          <button
            onClick={handleSaveAll}
            disabled={loading}
            className="w-full md:w-auto bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-3 rounded-xl shadow-lg transition text-xs flex items-center justify-center space-x-2 cursor-pointer"
          >
            <CheckCircle2 className="h-4 w-4" />
            <span>{loading ? 'Đang lưu...' : 'Lưu tất cả điểm & ảnh'}</span>
          </button>
        </div>

        {/* Bảng chấm điểm */}
        <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
          <table className="w-full text-left border-collapse">
            <thead>
              <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                <th className="p-4">STT</th>
                <th className="p-4">Họ và Tên Học Sinh</th>
                <th className="p-4">Tổ</th>
                <th className="p-4">Tổng Điểm</th>
                <th className="p-4">Xếp Loại</th>
                <th className="p-4 text-center">Ảnh Vi Phạm</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
              {students.map((st, index) => {
                const sData = scores[st.id] || { score: 100, eval: 'T', image_url: '' }
                return (
                  <tr key={st.id} className="hover:bg-gray-50/50">
                    <td className="p-4 font-bold">{index + 1}</td>
                    <td className="p-4 font-bold text-gray-900">{st.full_name}</td>
                    <td className="p-4">Tổ {st.team_number}</td>
                    <td className="p-4">
                      <input
                        type="number"
                        value={sData.score}
                        onChange={(e) => handleScoreChange(st.id, e.target.value)}
                        className="w-24 px-3 py-1.5 bg-gray-50 border border-gray-200 rounded-lg font-bold text-center text-purple-700 outline-none"
                      />
                    </td>
                    <td className="p-4 font-extrabold">
                      <span className={`px-2.5 py-1 rounded-lg text-xs ${
                        sData.eval === 'T' ? 'bg-green-100 text-green-700' :
                        sData.eval === 'K' ? 'bg-blue-100 text-blue-700' :
                        sData.eval === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'
                      }`}>
                        {sData.eval}
                      </span>
                    </td>
                    <td className="p-4 text-center">
                      <div className="flex items-center justify-center space-x-2">
                        {sData.image_url && (
                          <a
                            href={sData.image_url}
                            target="_blank"
                            rel="noopener noreferrer"
                            className="text-purple-600 hover:text-purple-800 font-bold flex items-center space-x-1 bg-purple-50 px-2.5 py-1.5 rounded-lg"
                          >
                            <ImageIcon className="h-3.5 w-3.5" />
                            <span>Xem</span>
                          </a>
                        )}
                        <label className="cursor-pointer bg-gray-100 hover:bg-gray-200 text-gray-700 px-3 py-1.5 rounded-lg font-bold flex items-center space-x-1 transition">
                          <Upload className="h-3.5 w-3.5" />
                          <span>Tải ảnh</span>
                          <input
                            type="file"
                            accept="image/*"
                            onChange={(e) => handleImageUpload(st.id, e)}
                            className="hidden"
                          />
                        </label>
                      </div>
                    </td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      </main>
    </div>
  )
}
