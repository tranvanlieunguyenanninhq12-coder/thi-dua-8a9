'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/utils/supabase/client'
import { Award, LogOut, Upload, CheckCircle2, ImageIcon, Trash2 } from 'lucide-react'

export default function ToTruongDashboard() {
  const TEN_LOP_CO_DINH = '8A9' 

  const [currentUser, setCurrentUser] = useState<any>(null)
  const [targetTeam, setTargetTeam] = useState<number>(0)
  const [students, setStudents] = useState<any[]>([])
  const [weeks, setWeeks] = useState<any[]>([])
  const [selectedWeek, setSelectedWeek] = useState<string>('')
  const [scores, setScores] = useState<{ [key: string]: { score: number; eval: string } }>({})
  const [teamImage, setTeamImage] = useState<string>('')
  const [loading, setLoading] = useState(false)
  const [uploading, setUploading] = useState(false)
  const [previewImage, setPreviewImage] = useState<string | null>(null)
  const [zoom, setZoom] = useState(1)
  const [rotation, setRotation] = useState(0)

  const router = useRouter()
  const supabase = createClient()

  useEffect(() => {
    const userStr = localStorage.getItem('currentUser')
    if (!userStr) { router.push('/login'); return; }
    const user = JSON.parse(userStr)
    if (!user.role.startsWith('to_truong_')) { router.push('/login'); return; }
    setCurrentUser(user)

    const tNum = user.team_number
    let mappedTarget = 1
    if (tNum === 1) mappedTarget = 2
    else if (tNum === 2) mappedTarget = 1
    else if (tNum === 3) mappedTarget = 4
    else if (tNum === 4) mappedTarget = 3

    setTargetTeam(mappedTarget)
    fetchInitialData(mappedTarget)
  }, [])

  const fetchInitialData = async (targetT: number) => {
    setLoading(true)
    const { data: weeksData } = await supabase.from('weeks').select('*')
    if (weeksData && weeksData.length > 0) {
      setWeeks(weeksData)
      setSelectedWeek(weeksData[0].id)
      fetchDataForWeek(targetT, weeksData[0].id)
    }
    setLoading(false)
  }

  const fetchDataForWeek = async (targetT: number, weekId: string) => {
    const { data: studentsData } = await supabase.from('students').select('*')
      .eq('class_name', TEN_LOP_CO_DINH).eq('team_number', targetT).order('full_name', { ascending: true })
    if (studentsData) setStudents(studentsData)

    const { data: scoresData } = await supabase.from('emulation_scores').select('*').eq('week_id', weekId)
    if (scoresData) {
      const scoreMap: any = {}
      scoresData.forEach((s: any) => { scoreMap[s.student_id] = { score: s.score, eval: s.evaluation } })
      setScores(scoreMap)
    }

    // Lấy ảnh không dùng class_name
    const { data: imgData } = await supabase.from('team_weekly_images').select('*').eq('week_id', weekId).eq('team_number', targetT)
    if (imgData && imgData.length > 0) {
        setTeamImage(imgData[0].image_url)
    } else {
        setTeamImage('')
    }
  }

  const handleWeekChange = (weekId: string) => {
    setSelectedWeek(weekId)
    fetchDataForWeek(targetTeam, weekId)
  }

  const calculateEvaluation = (score: number) => {
    if (score < 50) return 'Y'; if (score <= 64) return 'TB'; if (score <= 79) return 'K'; return 'T';
  }

  const handleScoreChange = (studentId: string, val: string) => {
    const num = parseFloat(val) || 0
    const ev = calculateEvaluation(num)
    setScores(prev => ({ ...prev, [studentId]: { score: num, eval: ev } }))
  }
const compressImage = (file: File): Promise<Blob> => {
    return new Promise((resolve, reject) => {
      const reader = new FileReader()
      reader.readAsDataURL(file)
      reader.onload = (event) => {
        const img = new Image()
        img.src = event.target?.result as string
        img.onload = () => {
          const canvas = document.createElement('canvas')
          let width = img.width; let height = img.height; const maxDimension = 1200
          if (width > height && width > maxDimension) { height = (height * maxDimension) / width; width = maxDimension } 
          else if (height > maxDimension) { width = (width * maxDimension) / height; height = maxDimension }
          canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d')
          ctx?.drawImage(img, 0, 0, width, height)
          canvas.toBlob((blob) => { if (blob) resolve(blob); else reject(new Error('Lỗi nén ảnh')) }, 'image/jpeg', 0.8)
        }
      }
      reader.onerror = (error) => reject(error)
    })
  }

  const handleTeamImageUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file || !selectedWeek) return
    setUploading(true)
    try {
      const compressedBlob = await compressImage(file)
      const fileName = `to_${targetTeam}_w_${selectedWeek}_${Date.now()}.jpg`
      const { error: uploadError } = await supabase.storage.from('thi-dua').upload(fileName, compressedBlob, { contentType: 'image/jpeg' })
      if (uploadError) throw uploadError
      const { data: publicUrlData } = supabase.storage.from('thi-dua').getPublicUrl(fileName)
      const publicUrl = publicUrlData.publicUrl
      
      // Xử lý lưu ảnh chuẩn mực không dùng thuộc tính class_name
      const { data: existing } = await supabase.from('team_weekly_images').select('id').eq('week_id', selectedWeek).eq('team_number', targetTeam).maybeSingle()
      
      if (existing) {
        const { error: updateError } = await supabase.from('team_weekly_images').update({ image_url: publicUrl }).eq('id', existing.id)
        if (updateError) throw updateError
      } else {
        const { error: insertError } = await supabase.from('team_weekly_images').insert([{ week_id: selectedWeek, team_number: targetTeam, image_url: publicUrl }])
        if (insertError) throw insertError
      }
      
      setTeamImage(publicUrl)
      alert(`Đã tải và nén ảnh minh chứng cho Tổ ${targetTeam} thành công!`)
    } catch (err: any) { alert('Lỗi tải ảnh tổ lên: ' + err.message) } 
    finally { setUploading(false) }
  }

  const handleDeleteTeamImage = async () => {
    if (!confirm(`Thầy/Cô có chắc chắn muốn thu hồi/xóa ảnh minh chứng của Tổ ${targetTeam} không?`)) return
    await supabase.from('team_weekly_images').delete().eq('week_id', selectedWeek).eq('team_number', targetTeam)
    setTeamImage('')
    alert('Đã thu hồi ảnh thành công!')
  }

  const handleSaveAll = async () => {
    if (!selectedWeek) return
    setLoading(true)
    for (const student of students) {
      const studentScore = scores[student.id] || { score: 100, eval: 'T' }
      const { data: existing } = await supabase.from('emulation_scores').select('id').eq('student_id', student.id).eq('week_id', selectedWeek).maybeSingle()
      if (existing) { 
        await supabase.from('emulation_scores').update({ score: studentScore.score, evaluation: studentScore.eval, updated_by: currentUser.username }).eq('id', existing.id) 
      } else { 
        await supabase.from('emulation_scores').insert([{ student_id: student.id, week_id: selectedWeek, score: studentScore.score, evaluation: studentScore.eval, updated_by: currentUser.username }]) 
      }
    }
    setLoading(false)
    alert('Đã lưu điểm và thông tin tổ thành công!')
  }
return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      <header className="bg-white border-b border-purple-100 shadow-xs px-6 py-4 flex justify-between items-center">
        <div className="flex items-center space-x-3">
          <div className="bg-purple-600 text-white p-2.5 rounded-2xl shadow-md"><Award className="h-6 w-6" /></div>
          <div><h1 className="text-lg font-black text-gray-900">Khu vực Chấm điểm - {TEN_LOP_CO_DINH}</h1><p className="text-xs text-purple-600 font-bold">{currentUser?.full_name} — Đang chấm cho <span className="text-red-600 uppercase font-extrabold">Tổ {targetTeam}</span></p></div>
        </div>
        <button onClick={() => { localStorage.removeItem('currentUser'); router.push('/login'); }} className="flex items-center space-x-2 bg-red-50 hover:bg-red-100 text-red-600 px-4 py-2.5 rounded-xl text-xs font-bold cursor-pointer transition shadow-sm"><LogOut className="h-4 w-4" /><span>Đăng xuất</span></button>
      </header>

      <main className="flex-1 max-w-6xl w-full mx-auto p-6 space-y-6">
        <div className="bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100 flex flex-col md:flex-row justify-between items-center gap-4">
          <div className="flex items-center space-x-3 w-full md:w-auto">
            <span className="text-xs font-extrabold text-gray-700">Chọn Tuần:</span>
            <select value={selectedWeek} onChange={(e) => handleWeekChange(e.target.value)} className="px-4 py-3 bg-purple-50 border border-purple-200 rounded-2xl font-bold text-purple-900 text-xs cursor-pointer outline-none focus:ring-2 focus:ring-purple-500/30 transition">
              {weeks.map((w) => <option key={w.id} value={w.id}>{w.week_name} ({w.period})</option>)}
            </select>
          </div>
          <button onClick={handleSaveAll} disabled={loading} className="w-full md:w-auto bg-gradient-to-r from-purple-600 to-indigo-600 hover:from-purple-700 hover:to-indigo-700 text-white font-extrabold px-8 py-3.5 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center space-x-2 text-sm">
            <CheckCircle2 className="h-5 w-5" /><span>{loading ? 'Đang lưu...' : 'Lưu tất cả điểm'}</span>
          </button>
        </div>

        <div className="bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-purple-100 flex flex-col md:flex-row items-center justify-between gap-4">
          <div><h3 className="text-base font-extrabold text-gray-900">Ảnh minh chứng vi phạm của Tổ {targetTeam}</h3><p className="text-xs text-gray-500 mt-1 font-medium">Tải lên ảnh sổ hoặc phiếu tổng hợp lỗi (Ảnh được tự động nén nhẹ, rõ nét).</p></div>
          <div className="flex items-center space-x-3 w-full md:w-auto justify-end">
            {teamImage ? (
              <div className="flex items-center space-x-2">
                <button onClick={() => { setZoom(1); setRotation(0); setPreviewImage(teamImage); }} className="text-purple-600 font-bold text-xs bg-purple-50 hover:bg-purple-100 px-4 py-3 rounded-2xl flex items-center space-x-1 transition cursor-pointer shadow-sm"><ImageIcon className="h-4 w-4" /><span>Xem trực tiếp</span></button>
                <button onClick={handleDeleteTeamImage} className="bg-red-50 hover:bg-red-100 text-red-600 font-bold p-3 rounded-2xl text-xs flex items-center space-x-1 transition cursor-pointer shadow-sm" title="Thu hồi ảnh"><Trash2 className="h-4 w-4" /></button>
                <label className="cursor-pointer bg-blue-600 hover:bg-blue-700 text-white px-5 py-3 rounded-2xl font-bold text-xs flex items-center space-x-1.5 transition shadow-md hover:-translate-y-0.5 transform"><Upload className="h-4 w-4" /><span>{uploading ? 'Đang nén...' : 'Tải lại ảnh'}</span><input type="file" accept="image/*" onChange={handleTeamImageUpload} disabled={uploading} className="hidden" /></label>
              </div>
            ) : (
              <label className="cursor-pointer bg-gradient-to-r from-purple-600 to-indigo-600 hover:from-purple-700 hover:to-indigo-700 text-white px-6 py-3 rounded-2xl font-bold text-xs flex items-center space-x-1.5 transition shadow-lg hover:-translate-y-0.5 transform"><Upload className="h-4 w-4" /><span>{uploading ? 'Đang nén...' : 'Tải ảnh tổ lên'}</span><input type="file" accept="image/*" onChange={handleTeamImageUpload} disabled={uploading} className="hidden" /></label>
            )}
          </div>
        </div>

        <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
          <table className="w-full text-left border-collapse">
            <thead><tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold"><th className="p-4">STT</th><th className="p-4">Họ và Tên Học Sinh</th><th className="p-4">Tổ</th><th className="p-4">Tổng Điểm</th><th className="p-4">Xếp Loại</th></tr></thead>
            <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
              {students.map((st, index) => {
                const sData = scores[st.id] || { score: 100, eval: 'T' }
                return (
                  <tr key={st.id} className="hover:bg-gray-50/50 transition-colors">
                    <td className="p-4 font-bold">{index + 1}</td><td className="p-4 font-bold text-gray-900">{st.full_name}</td><td className="p-4">Tổ {st.team_number}</td>
                    <td className="p-4"><input type="number" value={sData.score} onChange={(e) => handleScoreChange(st.id, e.target.value)} className="w-24 px-3 py-2 bg-white border border-gray-200 rounded-xl font-bold text-center text-purple-700 outline-none focus:ring-2 focus:ring-purple-500/30 transition shadow-sm" /></td>
                    <td className="p-4 font-extrabold"><span className={`px-3 py-1.5 rounded-xl text-xs ${sData.eval === 'T' ? 'bg-green-100 text-green-700' : sData.eval === 'K' ? 'bg-blue-100 text-blue-700' : sData.eval === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'}`}>{sData.eval}</span></td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      </main>

      {previewImage && (
        <div className="fixed inset-0 bg-black/90 z-50 flex flex-col items-center justify-center p-4 backdrop-blur-sm">
          <div className="absolute top-6 right-6 flex space-x-3 bg-gray-900/50 p-2 rounded-2xl backdrop-blur-md">
            <button onClick={() => setZoom(z => z + 0.2)} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Phóng to (+)</button>
            <button onClick={() => setZoom(z => Math.max(0.4, z - 0.2))} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Thu nhỏ (-)</button>
            <button onClick={() => setRotation(r => (r + 90) % 360)} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Xoay 90°</button>
            <button onClick={() => setPreviewImage(null)} className="bg-red-500 hover:bg-red-600 text-white px-6 py-2 rounded-xl text-xs font-extrabold cursor-pointer transition shadow-lg">Đóng (X)</button>
          </div>
          <div className="overflow-auto max-w-full max-h-[85vh] flex items-center justify-center p-2 mt-12">
            <img src={previewImage} alt="Minh chứng" style={{ transform: `scale(${zoom}) rotate(${rotation}deg)`, transition: 'transform 0.25s cubic-bezier(0.4, 0, 0.2, 1)' }} className="max-h-[75vh] object-contain rounded-xl shadow-2xl ring-4 ring-white/10" />
          </div>
        </div>
      )}
    </div>
  )
}
