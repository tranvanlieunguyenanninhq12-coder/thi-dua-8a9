'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/utils/supabase/client'
import { ShieldCheck, LogOut, Upload, ImageIcon, Plus } from 'lucide-react'

export default function PhoKyLuatDashboard() {
  const [currentUser, setCurrentUser] = useState<any>(null)
  const [records, setRecords] = useState<any[]>([])
  const [content, setContent] = useState('')
  const [teacherName, setTeacherName] = useState('')
  const [imageUrl, setImageUrl] = useState('')
  const [loading, setLoading] = useState(false)

  const router = useRouter()
  const supabase = createClient()

  useEffect(() => {
    const userStr = localStorage.getItem('currentUser')
    if (!userStr) {
      router.push('/login')
      return
    }
    const user = JSON.parse(userStr)
    if (user.role !== 'pho_ky_luat') {
      router.push('/login')
      return
    }
    setCurrentUser(user)
    fetchRecords()
  }, [])

  const fetchRecords = async () => {
    const { data } = await supabase.from('disciplinary_records').select('*').order('created_at', { ascending: false })
    if (data) setRecords(data)
  }

  const handleImageUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    setLoading(true)
    const fileName = `kl_${Date.now()}_${file.name}`
    const { error } = await supabase.storage.from('thi-dua').upload(fileName, file)

    if (error) {
      alert('Lỗi tải ảnh!')
      setLoading(false)
      return
    }

    const { data: publicUrlData } = supabase.storage.from('thi-dua').getPublicUrl(fileName)
    setImageUrl(publicUrlData.publicUrl)
    setLoading(false)
    alert('Tải ảnh sổ kỷ luật thành công!')
  }

  const handleAddRecord = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!content.trim()) return

    const { error } = await supabase.from('disciplinary_records').insert([
      { content, teacher_name: teacherName, image_url: imageUrl }
    ])

    if (error) {
      alert('Lỗi khi ghi nhận kỷ luật!')
    } else {
      alert('Đã ghi nhận và tải ảnh thành công!')
      setContent('')
      setTeacherName('')
      setImageUrl('')
      fetchRecords()
    }
  }

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      <header className="bg-white border-b border-purple-100 px-6 py-4 flex justify-between items-center shadow-xs">
        <div className="flex items-center space-x-3">
          <div className="bg-purple-600 text-white p-2.5 rounded-2xl">
            <ShieldCheck className="h-6 w-6" />
          </div>
          <div>
            <h1 className="text-lg font-black text-gray-900">Khu Vực Phó Kỷ Luật</h1>
            <p className="text-xs text-purple-600 font-bold">Xin chào, {currentUser?.full_name}</p>
          </div>
        </div>
        <button
          onClick={() => { localStorage.removeItem('currentUser'); router.push('/login'); }}
          className="flex items-center space-x-2 bg-red-50 text-red-600 px-4 py-2 rounded-xl text-xs font-bold cursor-pointer"
        >
          <LogOut className="h-4 w-4" />
          <span>Đăng xuất</span>
        </button>
      </header>

      <main className="flex-1 max-w-4xl w-full mx-auto p-6 space-y-6">
        <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4">
          <h3 className="text-base font-extrabold text-gray-900">Đăng Sổ Kỷ Luật Giáo Viên Bộ Môn</h3>
          <form onSubmit={handleAddRecord} className="space-y-4">
            <textarea
              placeholder="Nội dung vi phạm của học sinh ghi nhận từ giáo viên bộ môn..."
              value={content}
              onChange={(e) => setContent(e.target.value)}
              required
              className="w-full p-4 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none"
              rows={3}
            />
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <input
                type="text"
                placeholder="Tên giáo viên bộ môn ghi nhận"
                value={teacherName}
                onChange={(e) => setTeacherName(e.target.value)}
                className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold"
              />
              <label className="flex items-center justify-center space-x-2 bg-purple-50 hover:bg-purple-100 text-purple-700 px-4 py-3 rounded-xl border border-purple-200 cursor-pointer font-bold text-xs">
                <Upload className="h-4 w-4" />
                <span>{imageUrl ? 'Đã có ảnh đính kèm' : 'Tải ảnh sổ kỷ luật'}</span>
                <input type="file" accept="image/*" onChange={handleImageUpload} className="hidden" />
              </label>
            </div>
            <button
              type="submit"
              disabled={loading}
              className="w-full bg-purple-600 hover:bg-purple-700 text-white font-extrabold py-3 rounded-xl shadow transition text-xs cursor-pointer flex items-center justify-center space-x-2"
            >
              <Plus className="h-4 w-4" />
              <span>{loading ? 'Đang xử lý...' : 'Lưu bản ghi kỷ luật'}</span>
            </button>
          </form>
        </div>

        <div className="bg-white rounded-3xl shadow-sm border border-gray-100 p-6 space-y-4">
          <h3 className="text-base font-extrabold text-gray-900">Danh Sách Sổ Kỷ Luật Đã Ghi Nhận</h3>
          <div className="space-y-3">
            {records.map((r) => (
              <div key={r.id} className="p-4 bg-gray-50 rounded-2xl border border-gray-100 space-y-2">
                <p className="text-xs font-bold text-gray-900">{r.content}</p>
                <div className="flex justify-between items-center text-xs text-gray-500 font-medium">
                  <span>Giáo viên: <b>{r.teacher_name || 'Không rõ'}</b></span>
                  {r.image_url && (
                    <a href={r.image_url} target="_blank" rel="noopener noreferrer" className="text-purple-600 font-bold flex items-center space-x-1">
                      <ImageIcon className="h-3.5 w-3.5" />
                      <span>Xem ảnh</span>
                    </a>
                  )}
                </div>
              </div>
            ))}
          </div>
        </div>
      </main>
    </div>
  )
}
