'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/utils/supabase/client'
import { ShieldAlert, Users, Calendar, Award, BarChart3, Settings, LogOut, Trash2, FileSpreadsheet, ExternalLink, ImageIcon, BookOpen, Home, RefreshCw, Plus, CheckCircle2 } from 'lucide-react'

export default function AdminDashboard() {
  const TEN_LOP_CO_DINH = '8A9'

  const [currentTab, setCurrentTab] = useState('home')
  const [currentUser, setCurrentUser] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  
  const [usersList, setUsersList] = useState<any[]>([])
  const [newUsername, setNewUsername] = useState('')
  const [newPassword, setNewPassword] = useState('')
  const [newFullName, setNewFullName] = useState('')
  const [newRole, setNewRole] = useState('to_truong_1')

  const [selectedClass, setSelectedClass] = useState(TEN_LOP_CO_DINH)
  const [studentsList, setStudentsList] = useState<any[]>([])
  const [newStudentName, setNewStudentName] = useState('')
  const [newStudentCode, setNewStudentCode] = useState('')
  const [newStudentTeam, setNewStudentTeam] = useState(1)

  const [weeksList, setWeeksList] = useState<any[]>([])
  const [selectedWeek, setSelectedWeek] = useState('')
  const [scoresList, setScoresList] = useState<any[]>([])
  const [teamImages, setTeamImages] = useState<any[]>([])
  const [newWeekName, setNewWeekName] = useState('')
  const [newWeekSemester, setNewWeekSemester] = useState('HK1')
  const [newWeekPeriod, setNewWeekPeriod] = useState('GIỮA HỌC KỲ 1')

  const [selectedPeriod, setSelectedPeriod] = useState('GIỮA HỌC KỲ 1')
  const [disciplinaryList, setDisciplinaryList] = useState<any[]>([])
  const [resetPasswordInput, setResetPasswordInput] = useState('')
  
  const [previewImage, setPreviewImage] = useState<string | null>(null)
  const [zoom, setZoom] = useState(1)
  const [rotation, setRotation] = useState(0)

  const router = useRouter()
  const supabase = createClient()

  useEffect(() => {
    const userStr = localStorage.getItem('currentUser')
    if (!userStr) { router.push('/login'); return; }
    const user = JSON.parse(userStr)
    if (user.role !== 'admin') { router.push('/login'); return; }
    setCurrentUser(user)

    setSelectedClass(TEN_LOP_CO_DINH)
    fetchAllData(TEN_LOP_CO_DINH)
  }, [selectedWeek])

  const fetchAllData = async (className?: string) => {
    setLoading(true)
    const clsToFetch = className || TEN_LOP_CO_DINH
    
    const { data: users } = await supabase.from('users').select('*')
    if (users) setUsersList(users)

    const { data: students } = await supabase.from('students').select('*').eq('class_name', clsToFetch).order('full_name', { ascending: true })
    if (students) setStudentsList(students)

    const { data: weeks } = await supabase.from('weeks').select('*')
    if (weeks) {
      setWeeksList(weeks)
      if (weeks.length > 0 && !selectedWeek) setSelectedWeek(weeks[0].id)
    }

    const { data: allScores } = await supabase.from('emulation_scores').select('*')
    if (allScores) setScoresList(allScores)

    const currentWId = selectedWeek || (weeks && weeks.length > 0 ? weeks[0].id : '')
    if (currentWId) {
      const { data: imgData, error } = await supabase.from('team_weekly_images').select('*').eq('week_id', currentWId)
      if (error) { console.error("Lỗi tải ảnh:", error); }
      if (imgData && imgData.length > 0) {
        setTeamImages(imgData)
      } else {
        setTeamImages([])
      }
    }

    const { data: disc } = await supabase.from('disciplinary_records').select('*').order('created_at', { ascending: false })
    if (disc) setDisciplinaryList(disc)
    
    setLoading(false)
  }
const handleLogout = () => { localStorage.removeItem('currentUser'); router.push('/login'); }

  const handleAddUser = async (e: React.FormEvent) => {
    e.preventDefault()
    const teamNum = newRole.startsWith('to_truong') ? parseInt(newRole.replace('to_truong_', '')) : 0
    const { error } = await supabase.from('users').insert([{
      username: newUsername.trim().toUpperCase(), password: newPassword, full_name: newFullName, role: newRole, team_number: teamNum
    }])
    if (error) alert('Lỗi: Trùng tên tài khoản!')
    else { alert('Tạo thành công!'); setNewUsername(''); setNewPassword(''); setNewFullName(''); fetchAllData(); }
  }

  const handleDeleteUser = async (id: string) => {
    if (confirm('Xóa tài khoản này?')) { await supabase.from('users').delete().eq('id', id); fetchAllData(); }
  }

  const handleAddStudent = async (e: React.FormEvent) => {
    e.preventDefault()
    const { error } = await supabase.from('students').insert([{ 
      student_code: newStudentCode, full_name: newStudentName, class_name: TEN_LOP_CO_DINH, team_number: newStudentTeam 
    }])
    if (error) alert('Lỗi thêm học sinh!'); else { alert('Thêm thành công!'); setNewStudentName(''); setNewStudentCode(''); fetchAllData(); }
  }

  const handleDeleteStudent = async (id: string) => {
    if (confirm('Xóa học sinh này?')) { await supabase.from('students').delete().eq('id', id); fetchAllData(); }
  }

  const handleAddWeek = async (e: React.FormEvent) => {
    e.preventDefault()
    const { error } = await supabase.from('weeks').insert([{ week_name: newWeekName, semester: newWeekSemester, period: newWeekPeriod }])
    if (error) alert('Lỗi thêm tuần!'); else { alert('Thêm tuần thành công!'); setNewWeekName(''); fetchAllData(); }
  }

  const getEvaluation = (score: number) => {
    if (score < 50) return 'Y'; if (score <= 64) return 'TB'; if (score <= 79) return 'K'; return 'T';
  }

  const handleScoreUpdate = async (studentId: string, scoreVal: number) => {
    const evalVal = getEvaluation(scoreVal)
    const existing = scoresList.find(s => s.student_id === studentId && s.week_id === selectedWeek)
    
    if (existing) {
      await supabase.from('emulation_scores').update({ score: scoreVal, evaluation: evalVal }).eq('id', existing.id)
    } else {
      await supabase.from('emulation_scores').insert([{ student_id: studentId, week_id: selectedWeek, score: scoreVal, evaluation: evalVal }])
    }
    fetchAllData()
  }

  const handleDeleteTeamImage = async (imageId: string) => {
    if (!confirm('Thầy/Cô có chắc chắn muốn xóa ảnh này?')) return
    await supabase.from('team_weekly_images').delete().eq('id', imageId)
    fetchAllData(); alert('Đã xóa ảnh!')
  }

  const handleDeleteDisciplinary = async (id: string) => {
    if (!confirm('Xóa kỷ luật này?')) return
    await supabase.from('disciplinary_records').delete().eq('id', id)
    fetchAllData()
  }

  const exportToExcel = () => {
    let csv = "STT,Họ và Tên,Tổ,Tổng Điểm,Xếp Loại\n"
    studentsList.forEach((st, idx) => {
      const sc = scoresList.find(s => s.student_id === st.id && s.week_id === selectedWeek) || { score: 100, evaluation: 'T' }
      csv += `${idx + 1},"${st.full_name}",Tổ ${st.team_number},${sc.score},${sc.evaluation}\n`
    })
    const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' })
    const url = URL.createObjectURL(blob)
    const link = document.createElement("a")
    link.href = url
    link.setAttribute("download", `bao_cao_${TEN_LOP_CO_DINH}.csv`)
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
  }

  const handleResetSystem = async () => {
    if (resetPasswordInput !== 'luuvanglien0410') { alert('Sai mật khẩu Admin!'); return; }
    if (confirm('Xóa sạch dữ liệu cũ?')) {
      await supabase.from('emulation_scores').delete().neq('score', -999)
      await supabase.from('disciplinary_records').delete().neq('id', '00000000-0000-0000-0000-000000000000')
      await supabase.from('team_weekly_images').delete().neq('id', '00000000-0000-0000-0000-000000000000')
      alert('Đã làm mới hệ thống!')
      setResetPasswordInput('')
      router.push('/')
    }
  }
return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      <header className="bg-white border-b border-purple-100 px-6 py-4 flex justify-between items-center shadow-xs">
        <div className="flex items-center space-x-3">
          <div className="bg-purple-600 text-white p-2.5 rounded-2xl shadow-md"><ShieldAlert className="h-6 w-6" /></div>
          <div><h1 className="text-lg font-black text-gray-900">Quản Trị Viên</h1><p className="text-xs text-purple-600 font-bold">GVCN: TRẦN VĂN LIỆU</p></div>
        </div>
        <div className="flex items-center space-x-3">
          {currentTab !== 'home' && (
            <button onClick={() => setCurrentTab('home')} className="bg-purple-50 hover:bg-purple-100 text-purple-700 px-4 py-2.5 rounded-xl text-xs font-bold cursor-pointer transition"><Home className="h-4 w-4 inline mr-1" />Trang chủ</button>
          )}
          <div className="px-4 py-2.5 bg-purple-100 rounded-xl font-black text-purple-900 text-xs shadow-inner">Lớp: {TEN_LOP_CO_DINH}</div>
          <button onClick={handleLogout} className="bg-red-50 hover:bg-red-100 text-red-600 px-4 py-2.5 rounded-xl text-xs font-bold cursor-pointer transition shadow-sm"><LogOut className="h-4 w-4 inline mr-1" />Đăng xuất</button>
        </div>
      </header>

      <main className="flex-1 max-w-6xl w-full mx-auto p-6 space-y-6">
        {currentTab === 'home' && (
          <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
            {[
              { id: 'accounts', title: 'Quản lý Tài khoản', icon: Users, desc: 'Thêm, sửa, xóa tài khoản.' },
              { id: 'students', title: `Danh sách Học sinh ${TEN_LOP_CO_DINH}`, icon: Users, desc: 'Cập nhật danh sách 4 tổ.' },
              { id: 'weeks', title: 'Thời gian & Tuần', icon: Calendar, desc: 'Thêm tuần và đợt đánh giá.' },
              { id: 'results', title: 'Kết quả Thi đua Tuần', icon: Award, desc: 'Chấm điểm và xem ảnh.' },
              { id: 'periods', title: 'Thống kê theo Đợt', icon: BarChart3, desc: 'Tính điểm trung bình cộng.' },
              { id: 'saodo', title: 'Tra cứu Sao Đỏ', icon: ExternalLink, desc: 'Liên kết tra cứu trường.' },
              { id: 'teachers', title: 'Sổ kỷ luật GVBM', icon: BookOpen, desc: 'Xem vi phạm bộ môn.' },
              { id: 'settings', title: 'Cài đặt Dữ liệu', icon: Settings, desc: 'Làm mới dữ liệu hệ thống.' },
            ].map(c => {
              const Icon = c.icon
              return (
                <div key={c.id} onClick={() => setCurrentTab(c.id)} className="bg-white p-6 rounded-3xl shadow-sm border border-purple-50 hover:shadow-lg transition transform hover:-translate-y-1 cursor-pointer flex flex-col justify-between space-y-4 group">
                  <div className="bg-purple-50 text-purple-600 p-3 rounded-2xl w-fit group-hover:bg-purple-600 group-hover:text-white transition-colors"><Icon className="h-6 w-6" /></div>
                  <div><h3 className="font-extrabold text-gray-900">{c.title}</h3><p className="text-xs text-gray-500 font-medium mt-1">{c.desc}</p></div>
                </div>
              )
            })}
          </div>
        )}

        {currentTab === 'accounts' && (
          <div className="space-y-6 bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100">
            <h3 className="text-base font-extrabold text-gray-900">Cấp tài khoản mới</h3>
            <form onSubmit={handleAddUser} className="grid grid-cols-1 md:grid-cols-5 gap-4">
              <input type="text" placeholder="Mã TK" value={newUsername} onChange={e => setNewUsername(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold uppercase outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <input type="text" placeholder="Mật khẩu" value={newPassword} onChange={e => setNewPassword(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <input type="text" placeholder="Họ tên" value={newFullName} onChange={e => setNewFullName(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <select value={newRole} onChange={e => setNewRole(e.target.value)} className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none cursor-pointer focus:ring-2 focus:ring-purple-500/30 transition">
                <option value="to_truong_1">Tổ trưởng 1</option><option value="to_truong_2">Tổ trưởng 2</option><option value="to_truong_3">Tổ trưởng 3</option><option value="to_truong_4">Tổ trưởng 4</option><option value="pho_ky_luat">Phó kỷ luật</option>
              </select>
              <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-3 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center"><Plus className="h-4 w-4 mr-1"/>Tạo</button>
            </form>
            <div className="overflow-hidden rounded-2xl border border-gray-100 mt-6">
              <table className="w-full text-left border-collapse">
                <thead><tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold"><th className="p-4">STT</th><th className="p-4">Mã TK</th><th className="p-4">Họ tên</th><th className="p-4">Quyền</th><th className="p-4 text-center">Xóa</th></tr></thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {usersList.map((u, i) => (
                    <tr key={u.id} className="hover:bg-gray-50/50 transition-colors"><td className="p-4 font-bold">{i+1}</td><td className="p-4 font-extrabold text-purple-700">{u.username}</td><td className="p-4">{u.full_name}</td><td className="p-4 uppercase">{u.role}</td><td className="p-4 text-center">{u.username !== 'ADMIN' && (<button onClick={() => handleDeleteUser(u.id)} className="text-red-500 hover:text-red-700 hover:bg-red-50 p-2 rounded-xl transition cursor-pointer"><Trash2 className="h-4 w-4 mx-auto" /></button>)}</td></tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {currentTab === 'students' && (
          <div className="space-y-6 bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100">
            <h3 className="text-base font-extrabold text-gray-900">Thêm học sinh lớp {TEN_LOP_CO_DINH}</h3>
            <form onSubmit={handleAddStudent} className="grid grid-cols-1 md:grid-cols-4 gap-4">
              <input type="text" placeholder="Mã HS" value={newStudentCode} onChange={e => setNewStudentCode(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <input type="text" placeholder="Họ tên HS" value={newStudentName} onChange={e => setNewStudentName(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <select value={newStudentTeam} onChange={e => setNewStudentTeam(parseInt(e.target.value))} className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none cursor-pointer focus:ring-2 focus:ring-purple-500/30 transition"><option value={1}>Tổ 1</option><option value={2}>Tổ 2</option><option value={3}>Tổ 3</option><option value={4}>Tổ 4</option></select>
              <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-3 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center"><Plus className="h-4 w-4 mr-1"/>Thêm</button>
            </form>
            <div className="overflow-hidden rounded-2xl border border-gray-100 mt-6">
              <table className="w-full text-left border-collapse">
                <thead><tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold"><th className="p-4">STT</th><th className="p-4">Mã HS</th><th className="p-4">Họ tên</th><th className="p-4">Tổ</th><th className="p-4 text-center">Xóa</th></tr></thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, i) => (
                    <tr key={st.id} className="hover:bg-gray-50/50 transition-colors"><td className="p-4 font-bold">{i+1}</td><td className="p-4 font-extrabold text-purple-700">{st.student_code}</td><td className="p-4 font-bold text-gray-900">{st.full_name}</td><td className="p-4">Tổ {st.team_number}</td><td className="p-4 text-center"><button onClick={() => handleDeleteStudent(st.id)} className="text-red-500 hover:text-red-700 hover:bg-red-50 p-2 rounded-xl transition cursor-pointer"><Trash2 className="h-4 w-4 mx-auto" /></button></td></tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {currentTab === 'weeks' && (
          <div className="space-y-6 bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100">
            <h3 className="text-base font-extrabold text-gray-900">Thêm Tuần & Thời Gian</h3>
            <form onSubmit={handleAddWeek} className="grid grid-cols-1 md:grid-cols-4 gap-4">
              <input type="text" placeholder="VD: Tuần 1 - 07/09/2026" value={newWeekName} onChange={e => setNewWeekName(e.target.value)} required className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:ring-2 focus:ring-purple-500/30 transition" />
              <select value={newWeekSemester} onChange={e => setNewWeekSemester(e.target.value)} className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none cursor-pointer focus:ring-2 focus:ring-purple-500/30 transition"><option value="HK1">HK1</option><option value="HK2">HK2</option></select>
              <select value={newWeekPeriod} onChange={e => setNewWeekPeriod(e.target.value)} className="px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none cursor-pointer focus:ring-2 focus:ring-purple-500/30 transition"><option value="GIỮA HỌC KỲ 1">GIỮA HỌC KỲ 1</option><option value="HỌC KỲ 1">HỌC KỲ 1</option><option value="GIỮA HỌC KỲ 2">GIỮA HỌC KỲ 2</option><option value="HỌC KỲ 2">HỌC KỲ 2</option></select>
              <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-3 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center"><Plus className="h-4 w-4 mr-1"/>Tạo tuần</button>
            </form>
            <div className="overflow-hidden rounded-2xl border border-gray-100 mt-6">
              <table className="w-full text-left border-collapse">
                <thead><tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold"><th className="p-4">STT</th><th className="p-4">Tên Tuần</th><th className="p-4">Học Kỳ</th><th className="p-4">Đợt</th><th className="p-4 text-center">Xóa</th></tr></thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {weeksList.map((w, i) => (
                    <tr key={w.id} className="hover:bg-gray-50/50 transition-colors"><td className="p-4 font-bold">{i+1}</td><td className="p-4 font-extrabold text-purple-700">{w.week_name}</td><td className="p-4">{w.semester}</td><td className="p-4 font-bold">{w.period}</td><td className="p-4 text-center"><button onClick={async () => { if(confirm('Xóa tuần?')){ await supabase.from('weeks').delete().eq('id', w.id); fetchAllData(); } }} className="text-red-500 hover:text-red-700 hover:bg-red-50 p-2 rounded-xl transition cursor-pointer"><Trash2 className="h-4 w-4 mx-auto" /></button></td></tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {currentTab === 'results' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 flex flex-col md:flex-row justify-between items-center gap-4">
              <div className="flex items-center space-x-3 w-full md:w-auto">
                <span className="text-xs font-extrabold text-gray-700">Chọn Tuần:</span>
                <select value={selectedWeek} onChange={e => setSelectedWeek(e.target.value)} className="px-4 py-2.5 bg-purple-50 border border-purple-200 rounded-xl font-bold text-purple-900 text-xs cursor-pointer outline-none w-full md:w-auto">
                  {weeksList.map(w => (<option key={w.id} value={w.id}>{w.week_name} ({w.period})</option>))}
                </select>
                <button onClick={() => fetchAllData()} className="bg-blue-50 hover:bg-blue-100 text-blue-700 font-extrabold px-4 py-2.5 rounded-xl text-xs transition cursor-pointer flex items-center border border-blue-200 shadow-sm">
                  <RefreshCw className={`h-3 w-3 mr-1 ${loading ? 'animate-spin' : ''}`} /> Đồng bộ
                </button>
              </div>
              <button onClick={exportToExcel} className="w-full md:w-auto bg-green-600 hover:bg-green-700 text-white font-extrabold px-6 py-3 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center space-x-2 text-xs">
                <FileSpreadsheet className="h-4 w-4 inline" /><span>Xuất Excel Báo Cáo</span>
              </button>
            </div>
            
            <div className="bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100">
              <h3 className="text-base font-extrabold text-gray-900 mb-4">Thư viện ảnh minh chứng</h3>
              {teamImages.length > 0 ? (
                <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                  {teamImages.map((img) => (
                    <div key={img.id || Math.random()} className="bg-gray-50 p-3 rounded-2xl border border-gray-200 shadow-sm flex flex-col items-center">
                      <div className="w-full h-32 mb-3 overflow-hidden rounded-xl border border-gray-100 bg-white">
                         <img src={img.image_url} alt="Minh chứng" className="w-full h-full object-cover cursor-pointer hover:scale-110 transition duration-300" onClick={() => { setZoom(1); setRotation(0); setPreviewImage(img.image_url); }} />
                      </div>
                      <span className="text-[11px] font-extrabold text-gray-700 mb-3 bg-white px-3 py-1.5 rounded-full shadow-xs border border-gray-200">
                        Nguồn tải lên: Tổ {img.team_number || img.target_team || 'Không rõ'}
                      </span>
                      <div className="flex w-full space-x-2">
                        <button onClick={() => { setZoom(1); setRotation(0); setPreviewImage(img.image_url); }} className="flex-1 bg-purple-100 hover:bg-purple-200 text-purple-700 font-bold py-2 rounded-xl text-[11px] transition cursor-pointer">Xem</button>
                        <button onClick={() => handleDeleteTeamImage(img.id)} className="flex-1 bg-red-100 hover:bg-red-200 text-red-600 font-bold py-2 rounded-xl text-[11px] transition cursor-pointer">Xóa</button>
                      </div>
                    </div>
                  ))}
                </div>
              ) : (
                <div className="text-center py-10 bg-gray-50 rounded-2xl border border-dashed border-gray-300">
                  <p className="text-xs font-bold text-gray-500">Chưa có ảnh minh chứng nào được tải lên trong tuần này.</p>
                </div>
              )}
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th><th className="p-4">Họ tên</th><th className="p-4">Tổ</th><th className="p-4">Điểm</th><th className="p-4">Xếp loại</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, i) => {
                    const sc = scoresList.find(s => s.student_id === st.id && s.week_id === selectedWeek) || { score: 100, evaluation: 'T' }
                    return (
                      <tr key={st.id} className="hover:bg-gray-50/50 transition-colors">
                        <td className="p-4 font-bold">{i+1}</td>
                        <td className="p-4 font-bold text-gray-900">{st.full_name}</td>
                        <td className="p-4">Tổ {st.team_number}</td>
                        <td className="p-4"><input type="number" value={sc.score} onChange={e => handleScoreUpdate(st.id, parseFloat(e.target.value) || 0)} className="w-24 px-3 py-1.5 bg-gray-50 border border-gray-200 rounded-lg font-bold text-center text-purple-700 outline-none focus:ring-2 focus:ring-purple-500/30" /></td>
                        <td className="p-4 font-extrabold"><span className={`px-2.5 py-1 rounded-lg text-xs ${sc.evaluation === 'T' ? 'bg-green-100 text-green-700' : sc.evaluation === 'K' ? 'bg-blue-100 text-blue-700' : sc.evaluation === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'}`}>{sc.evaluation}</span></td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {currentTab === 'periods' && (
          <div className="space-y-6 bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100">
            <div className="flex items-center space-x-3 w-full md:w-auto pb-2">
              <span className="text-xs font-extrabold text-gray-700">Chọn Đợt Thống Kê:</span>
              <select value={selectedPeriod} onChange={e => setSelectedPeriod(e.target.value)} className="px-4 py-2.5 bg-purple-50 border border-purple-200 rounded-xl font-bold text-purple-900 text-xs outline-none cursor-pointer">
                <option value="GIỮA HỌC KỲ 1">GIỮA HỌC KỲ 1</option><option value="HỌC KỲ 1">HỌC KỲ 1</option><option value="GIỮA HỌC KỲ 2">GIỮA HỌC KỲ 2</option><option value="HỌC KỲ 2">HỌC KỲ 2</option><option value="CẢ NĂM">CẢ NĂM</option>
              </select>
            </div>
            <div className="overflow-hidden rounded-2xl border border-gray-100">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50/80 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th><th className="p-4">Họ tên</th><th className="p-4">Tổ</th><th className="p-4">Chi tiết điểm</th><th className="p-4">Điểm TB</th><th className="p-4">Xếp loại</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, i) => {
                    const filteredWeeks = weeksList.filter(w => {
                      const pUp = (w.period || '').toUpperCase(); const nUp = (w.week_name || '').toUpperCase();
                      if (selectedPeriod === 'GIỮA HỌC KỲ 1') return pUp.includes('GIỮA HỌC KỲ 1') || nUp.includes('GIỮA HỌC KỲ 1');
                      if (selectedPeriod === 'HỌC KỲ 1') return w.semester === 'HK1' || pUp.includes('HỌC KỲ 1') || nUp.includes('HỌC KỲ 1');
                      if (selectedPeriod === 'GIỮA HỌC KỲ 2') return pUp.includes('GIỮA HỌC KỲ 2') || nUp.includes('GIỮA HỌC KỲ 2');
                      if (selectedPeriod === 'HỌC KỲ 2') return w.semester === 'HK2' || pUp.includes('HỌC KỲ 2') || nUp.includes('HỌC KỲ 2');
                      if (selectedPeriod === 'CẢ NĂM') return true;
                      return false;
                    });
                    const weekIds = filteredWeeks.map(w => w.id)
                    const studentScoresInPeriod = weeksList.map(w => {
                      if (!weekIds.includes(w.id)) return null;
                      const found = scoresList.find(s => s.student_id === st.id && s.week_id === w.id);
                      return { weekName: w.week_name, score: found ? found.score : 100 };
                    }).filter(Boolean) as { weekName: string; score: number }[];

                    const avgScore = studentScoresInPeriod.length > 0 ? Math.round(studentScoresInPeriod.reduce((acc, curr) => acc + curr.score, 0) / studentScoresInPeriod.length) : 100
                    const evalPeriod = getEvaluation(avgScore)
                    const badgeCls = evalPeriod === 'T' ? 'bg-green-100 text-green-700' : evalPeriod === 'K' ? 'bg-blue-100 text-blue-700' : evalPeriod === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'

                    return (
                      <tr key={st.id} className="hover:bg-gray-50/50 transition-colors">
                        <td className="p-4 font-bold">{i+1}</td>
                        <td className="p-4 font-bold text-gray-900">{st.full_name}</td>
                        <td className="p-4">Tổ {st.team_number}</td>
                        <td className="p-4 text-gray-500 text-[11px] leading-relaxed">{studentScoresInPeriod.map(item => `${item.weekName}: ${item.score}đ`).join(' | ')}</td>
                        <td className="p-4 font-extrabold text-purple-700 text-sm">{avgScore}</td>
                        <td className="p-4 font-extrabold"><span className={`px-2.5 py-1 rounded-lg text-xs ${badgeCls}`}>{evalPeriod}</span></td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {currentTab === 'saodo' && (
          <div className="bg-white p-12 rounded-3xl shadow-sm border border-gray-100 text-center space-y-6">
            <div className="bg-purple-50 text-purple-600 p-4 rounded-full w-fit mx-auto"><ExternalLink className="h-10 w-10" /></div>
            <h3 className="text-xl font-black text-gray-900">Hệ Thống Tra Cứu Vi Phạm Nề Nếp</h3>
            <a href="https://thiduanamhoc2627.netlify.app/" target="_blank" rel="noopener noreferrer" className="inline-flex items-center space-x-2 bg-gradient-to-r from-purple-600 to-indigo-600 hover:from-purple-700 hover:to-indigo-700 text-white font-extrabold px-8 py-4 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 text-sm"><span>Mở trang tra cứu Sao Đỏ</span><ExternalLink className="h-4 w-4" /></a>
          </div>
        )}

        {currentTab === 'teachers' && (
          <div className="bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100 space-y-6">
            <h3 className="text-base font-extrabold text-gray-900">Sổ Kỷ Luật Giáo Viên Bộ Môn</h3>
            <div className="space-y-4">
              {disciplinaryList.map(d => (
                <div key={d.id} className="p-5 bg-gray-50 border border-gray-200 rounded-2xl flex flex-col md:flex-row justify-between items-start md:items-center gap-4 hover:shadow-sm transition">
                  <div className="space-y-1.5">
                    <p className="text-sm font-bold text-gray-900">{d.content}</p>
                    <p className="text-xs text-gray-500 font-medium">Giáo viên ghi nhận: <b className="text-gray-700">{d.teacher_name || 'Không rõ'}</b></p>
                  </div>
                  <div className="flex items-center space-x-3 w-full md:w-auto justify-end">
                    {d.image_url && (
                      <button onClick={() => { setZoom(1); setRotation(0); setPreviewImage(d.image_url); }} className="text-purple-600 text-xs bg-purple-50 hover:bg-purple-100 px-4 py-2.5 rounded-xl font-bold cursor-pointer transition flex items-center"><ImageIcon className="h-3.5 w-3.5 mr-1.5"/>Xem ảnh</button>
                    )}
                    <button onClick={() => handleDeleteDisciplinary(d.id)} className="bg-red-50 hover:bg-red-100 text-red-600 p-2.5 rounded-xl cursor-pointer transition"><Trash2 className="h-4 w-4" /></button>
                  </div>
                </div>
              ))}
              {disciplinaryList.length === 0 && (
                <div className="text-center py-10 border border-dashed border-gray-300 rounded-2xl">
                  <p className="text-xs font-bold text-gray-500">Chưa có bản ghi kỷ luật nào.</p>
                </div>
              )}
            </div>
          </div>
        )}

        {currentTab === 'settings' && (
          <div className="bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-gray-100 space-y-4 max-w-xl">
            <h3 className="text-base font-extrabold text-red-600 flex items-center"><ShieldAlert className="h-5 w-5 mr-2"/>Xóa & Khôi Phục Hệ Thống</h3>
            <p className="text-xs font-medium text-gray-500 pb-2">Hành động này sẽ xóa toàn bộ điểm số, ảnh minh chứng và sổ kỷ luật. Vui lòng nhập mật khẩu Admin để xác nhận.</p>
            <input type="password" placeholder="Nhập mật khẩu Admin..." value={resetPasswordInput} onChange={e => setResetPasswordInput(e.target.value)} className="w-full px-4 py-3 bg-gray-50 border border-gray-200 rounded-2xl text-xs font-semibold outline-none focus:border-red-400 focus:ring-2 focus:ring-red-100 transition" />
            <button onClick={handleResetSystem} className="w-full bg-red-600 hover:bg-red-700 text-white font-extrabold py-3.5 rounded-2xl shadow-lg transition transform hover:-translate-y-0.5 cursor-pointer flex items-center justify-center"><Trash2 className="h-4 w-4 mr-2"/>Xác Nhận Xóa Dữ Liệu Cũ</button>
          </div>
        )}
      </main>

      {previewImage && (
        <div className="fixed inset-0 bg-black/90 z-50 flex flex-col items-center justify-center p-4 backdrop-blur-sm">
          <div className="absolute top-6 right-6 flex space-x-3 bg-gray-900/50 p-2 rounded-2xl backdrop-blur-md">
            <button onClick={() => setZoom(z => z + 0.2)} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Phóng to (+)</button>
            <button onClick={() => setZoom(z => Math.max(0.4, z - 0.2))} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Thu nhỏ (-)</button>
            <button onClick={() => setRotation(r => (r + 90) % 360)} className="bg-white/10 hover:bg-white/20 text-white px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">Xoay 90°</button>
            <button onClick={() => setPreviewImage(null)} className="bg-red-500 hover:bg-red-600 text-white px-6 py-2 rounded-xl text-xs font-extrabold cursor-pointer transition shadow-lg">Đóng (X)</button>
          </div>
          <div className="overflow-auto max-w-full max-h-[85vh] flex items-center justify-center p-2 mt-12">
            <img src={previewImage} alt="Xem chi tiết" style={{ transform: `scale(${zoom}) rotate(${rotation}deg)`, transition: 'transform 0.25s cubic-bezier(0.4, 0, 0.2, 1)' }} className="max-h-[75vh] object-contain rounded-xl shadow-2xl ring-4 ring-white/10" />
          </div>
        </div>
      )}
    </div>
  )
}
