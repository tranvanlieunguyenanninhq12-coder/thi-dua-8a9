'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/utils/supabase/client'
import { 
  ShieldAlert, Users, Calendar, Award, BarChart3, Settings, 
  LogOut, Plus, Trash2, FileSpreadsheet, ExternalLink, ImageIcon, BookOpen, Home 
} from 'lucide-react'

export default function AdminDashboard() {
  const [currentTab, setCurrentTab] = useState('home')
  const [currentUser, setCurrentUser] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  
  const [usersList, setUsersList] = useState<any[]>([])
  const [newUsername, setNewUsername] = useState('')
  const [newPassword, setNewPassword] = useState('')
  const [newFullName, setNewFullName] = useState('')
  const [newRole, setNewRole] = useState('to_truong_1')

  const [selectedClass, setSelectedClass] = useState('8A1')
  const [studentsList, setStudentsList] = useState<any[]>([])
  const [newStudentName, setNewStudentName] = useState('')
  const [newStudentCode, setNewStudentCode] = useState('')
  const [newStudentTeam, setNewStudentTeam] = useState(1)

  const [weeksList, setWeeksList] = useState<any[]>([])
  const [selectedWeek, setSelectedWeek] = useState('')
  const [scoresList, setScoresList] = useState<any[]>([])
  const [teamImages, setTeamImages] = useState<{ [key: number]: string }>({})
  const [newWeekName, setNewWeekName] = useState('')
  const [newWeekSemester, setNewWeekSemester] = useState('HK1')
  const [newWeekPeriod, setNewWeekPeriod] = useState('GIỮA HỌC KỲ 1')

  const [selectedPeriod, setSelectedPeriod] = useState('GIỮA HỌC KỲ 1')
  const [disciplinaryList, setDisciplinaryList] = useState<any[]>([])
  const [resetPasswordInput, setResetPasswordInput] = useState('')

  // State cho Modal xem ảnh trực tiếp (Phóng to, Thu nhỏ, Xoay)
  const [previewImage, setPreviewImage] = useState<string | null>(null)
  const [zoom, setZoom] = useState(1)
  const [rotation, setRotation] = useState(0)

  const router = useRouter()
  const supabase = createClient()

  useEffect(() => {
    const userStr = localStorage.getItem('currentUser')
    if (!userStr) {
      router.push('/login')
      return
    }
    const user = JSON.parse(userStr)
    if (user.role !== 'admin') {
      router.push('/login')
      return
    }
    setCurrentUser(user)
    fetchAllData()
  }, [selectedClass, selectedWeek])

  const fetchAllData = async () => {
    setLoading(true)
    const { data: users } = await supabase.from('users').select('*')
    if (users) setUsersList(users)

    const { data: students } = await supabase.from('students').select('*').eq('class_name', selectedClass).order('full_name', { ascending: true })
    if (students) setStudentsList(students)

    const { data: weeks } = await supabase.from('weeks').select('*')
    if (weeks) {
      setWeeksList(weeks)
      if (weeks.length > 0 && !selectedWeek) setSelectedWeek(weeks[0].id)
    }

    if (selectedWeek) {
      const { data: scores } = await supabase.from('emulation_scores').select('*').eq('week_id', selectedWeek)
      if (scores) setScoresList(scores)

      const { data: imgData } = await supabase.from('team_weekly_images').select('*').eq('week_id', selectedWeek)
      if (imgData) {
        const imgMap: any = {}
        imgData.forEach((item: any) => {
          imgMap[item.team_number] = item.image_url
        })
        setTeamImages(imgMap)
      } else {
        setTeamImages({})
      }
    }

    const { data: disc } = await supabase.from('disciplinary_records').select('*').order('created_at', { ascending: false })
    if (disc) setDisciplinaryList(disc)

    setLoading(false)
  }

  const handleLogout = () => {
    localStorage.removeItem('currentUser')
    router.push('/login')
  }

  const handleAddUser = async (e: React.FormEvent) => {
    e.preventDefault()
    const { error } = await supabase.from('users').insert([
      {
        username: newUsername.trim().toUpperCase(),
        password: newPassword,
        full_name: newFullName,
        role: newRole,
        team_number: newRole.startsWith('to_truong') ? parseInt(newRole.replace('to_truong_', '')) : 0
      }
    ])
    if (error) alert('Lỗi: Trùng tên tài khoản!')
    else {
      alert('Tạo tài khoản thành công!')
      setNewUsername(''); setNewPassword(''); setNewFullName('')
      fetchAllData()
    }
  }

  const handleDeleteUser = async (id: string) => {
    if (confirm('Xóa tài khoản này?')) {
      await supabase.from('users').delete().eq('id', id)
      fetchAllData()
    }
  }

  const handleAddStudent = async (e: React.FormEvent) => {
    e.preventDefault()
    const { error } = await supabase.from('students').insert([
      { student_code: newStudentCode, full_name: newStudentName, class_name: selectedClass, team_number: newStudentTeam }
    ])
    if (error) alert('Lỗi thêm học sinh!')
    else {
      alert('Thêm học sinh thành công!')
      setNewStudentName(''); setNewStudentCode('')
      fetchAllData()
    }
  }

  const handleDeleteStudent = async (id: string) => {
    if (confirm('Xóa học sinh này?')) {
      await supabase.from('students').delete().eq('id', id)
      fetchAllData()
    }
  }

  const handleAddWeek = async (e: React.FormEvent) => {
    e.preventDefault()
    const { error } = await supabase.from('weeks').insert([
      { week_name: newWeekName, semester: newWeekSemester, period: newWeekPeriod }
    ])
    if (error) alert('Lỗi thêm tuần!')
    else {
      alert('Thêm tuần thành công!')
      setNewWeekName('')
      fetchAllData()
    }
  }

  const getEvaluation = (score: number) => {
    if (score < 50) return 'Y'
    if (score <= 64) return 'TB'
    if (score <= 79) return 'K'
    return 'T'
  }

  const handleScoreUpdate = async (studentId: string, scoreVal: number) => {
    const evalVal = getEvaluation(scoreVal)
    const existing = scoresList.find(s => s.student_id === studentId)

    if (existing) {
      await supabase.from('emulation_scores').update({ score: scoreVal, evaluation: evalVal }).eq('id', existing.id)
    } else {
      await supabase.from('emulation_scores').insert([{ student_id: studentId, week_id: selectedWeek, score: scoreVal, evaluation: evalVal }])
    }
    fetchAllData()
  }

  const handleDeleteTeamImage = async (teamNum: number) => {
    if (!confirm(`Admin có chắc chắn muốn xóa/thu hồi ảnh minh chứng của Tổ ${teamNum} không?`)) return
    await supabase.from('team_weekly_images').delete().eq('week_id', selectedWeek).eq('team_number', teamNum)
    fetchAllData()
    alert('Đã xóa ảnh của tổ thành công!')
  }

  const handleDeleteDisciplinary = async (id: string) => {
    if (!confirm('Admin có chắc chắn muốn xóa bản ghi kỷ luật này không?')) return
    await supabase.from('disciplinary_records').delete().eq('id', id)
    fetchAllData()
  }

  const exportToExcel = () => {
    let csvContent = "data:text/csv;charset=utf-8,STT,Họ và Tên,Tổ,Tổng Điểm,Xếp Loại\n"
    studentsList.forEach((st, idx) => {
      const sc = scoresList.find(s => s.student_id === st.id) || { score: 100, evaluation: 'T' }
      csvContent += `${idx + 1},"${st.full_name}",Tổ ${st.team_number},${sc.score},${sc.evaluation}\n`
    })
    const encodedUri = encodeURI(csvContent)
    const link = document.createElement("a")
    link.setAttribute("href", encodedUri)
    link.setAttribute("download", `bao_cao_thi_dua_${selectedClass}.csv`)
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
  }

  const handleResetSystem = async () => {
    if (resetPasswordInput !== 'luuvanglien0410') {
      alert('Mật khẩu Admin không chính xác!')
      return
    }
    if (confirm('CẢNH BÁO: Xóa sạch toàn bộ điểm số và ảnh cũ sang năm học mới?')) {
      await supabase.from('emulation_scores').delete().neq('score', -999)
      await supabase.from('disciplinary_records').delete().neq('id', '00000000-0000-0000-0000-000000000000')
      await supabase.from('team_weekly_images').delete().neq('id', '00000000-0000-0000-0000-000000000000')
      alert('Đã làm mới dữ liệu hệ thống thành công!')
      setResetPasswordInput('')
      fetchAllData()
    }
  }

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      {/* Header */}
      <header className="bg-white border-b border-purple-100 shadow-xs px-4 md:px-6 py-4 flex flex-col md:flex-row justify-between items-center gap-4">
        <div className="flex items-center space-x-3 w-full md:w-auto justify-between md:justify-start">
          <div className="flex items-center space-x-3">
            <div className="bg-purple-600 text-white p-2.5 rounded-2xl shadow-md">
              <ShieldAlert className="h-6 w-6" />
            </div>
            <div>
              <h1 className="text-base md:text-lg font-black text-gray-900">Quản Trị Viên Hệ Thống Thi Đua</h1>
              <p className="text-xs text-purple-600 font-bold">Giáo viên chủ nhiệm: TRẦN VĂN LIỆU</p>
            </div>
          </div>
        </div>

        <div className="flex items-center space-x-3 w-full md:w-auto justify-end">
          {currentTab !== 'home' && (
            <button 
              onClick={() => setCurrentTab('home')} 
              className="flex items-center space-x-1 bg-purple-50 hover:bg-purple-100 text-purple-700 px-4 py-2 rounded-xl text-xs font-extrabold cursor-pointer transition"
            >
              <Home className="h-4 w-4" />
              <span>Trang chủ</span>
            </button>
          )}
          <select 
            value={selectedClass} 
            onChange={(e) => setSelectedClass(e.target.value)}
            className="px-3 py-2 bg-purple-50 border border-purple-200 rounded-xl font-extrabold text-purple-900 text-xs cursor-pointer"
          >
            {Array.from({ length: 20 }, (_, i) => `8A${i + 1}`).map(cls => (
              <option key={cls} value={cls}>Lớp {cls}</option>
            ))}
          </select>
          <button onClick={handleLogout} className="flex items-center space-x-1 bg-red-50 hover:bg-red-100 text-red-600 px-4 py-2 rounded-xl text-xs font-bold cursor-pointer transition">
            <LogOut className="h-4 w-4" />
            <span>Đăng xuất</span>
          </button>
        </div>
      </header>

      {/* Main Container */}
      <main className="flex-1 max-w-7xl w-full mx-auto p-4 md:p-6 space-y-6">
        
        {/* TRANG CHỦ DẠNG THẺ (DASHBOARD CARDS) */}
        {currentTab === 'home' && (
          <div className="space-y-6">
            <div className="bg-gradient-to-r from-purple-600 to-indigo-600 rounded-3xl p-6 md:p-8 text-white shadow-xl space-y-2">
              <h2 className="text-xl md:text-2xl font-black">Bảng Điều Khiển Quản Trị Hệ Thống</h2>
              <p className="text-xs md:text-sm text-purple-100 font-medium">
                Quản lý toàn diện tài khoản, danh sách lớp, tuần thi đua, thống kê và kỷ luật giáo viên bộ môn tại đây.
              </p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              {[
                { id: 'accounts', title: 'Quản lý Tài khoản', desc: 'Thêm, sửa, xóa tài khoản Tổ trưởng, Phó kỷ luật, Học sinh.', icon: Users, badge: 'Tài khoản' },
                { id: 'students', title: 'Danh sách Học sinh', desc: `Cập nhật danh sách học sinh theo 4 tổ cho lớp ${selectedClass}.`, icon: Users, badge: 'Lớp học' },
                { id: 'weeks', title: 'Thời gian & Tuần', desc: 'Thêm tuần kèm khoảng ngày tháng và phân loại đợt đánh giá.', icon: Calendar, badge: 'Thời gian' },
                { id: 'results', title: 'Kết quả Thi đua Tuần', desc: 'Chấm điểm, tự động xếp loại và xem ảnh minh chứng 4 tổ.', icon: Award, badge: 'Thi đua' },
                { id: 'periods', title: 'Thống kê theo Đợt', desc: 'Tính điểm trung bình cộng Giữa kỳ, Cuối kỳ và Cả năm.', icon: BarChart3, badge: 'Thống kê' },
                { id: 'saodo', title: 'Tra cứu Sao Đỏ', desc: 'Liên kết trực tiếp đến hệ thống tra cứu vi phạm của trường.', icon: ExternalLink, badge: 'Tra cứu' },
                { id: 'teachers', title: 'Sổ kỷ luật GVBM', desc: 'Xem lại các ghi nhận vi phạm từ giáo viên bộ môn.', icon: BookOpen, badge: 'Kỷ luật' },
                { id: 'settings', title: 'Cài đặt Năm mới', desc: 'Làm mới dữ liệu hệ thống bảo mật bằng mật khẩu Admin.', icon: Settings, badge: 'Hệ thống' },
              ].map((card) => {
                const Icon = card.icon
                return (
                  <div
                    key={card.id}
                    onClick={() => setCurrentTab(card.id)}
                    className="bg-white p-6 rounded-3xl shadow-sm hover:shadow-xl border border-purple-100 transition-all duration-300 cursor-pointer flex flex-col justify-between space-y-4 group"
                  >
                    <div className="flex justify-between items-start">
                      <div className="bg-purple-50 group-hover:bg-purple-600 text-purple-600 group-hover:text-white p-3 rounded-2xl transition-colors duration-300">
                        <Icon className="h-6 w-6" />
                      </div>
                      <span className="text-[10px] font-extrabold px-3 py-1 bg-purple-50 text-purple-700 rounded-full uppercase tracking-wider">
                        {card.badge}
                      </span>
                    </div>
                    <div className="space-y-1">
                      <h3 className="text-base font-black text-gray-900 group-hover:text-purple-600 transition-colors">
                        {card.title}
                      </h3>
                      <p className="text-xs text-gray-500 font-medium leading-relaxed">
                        {card.desc}
                      </p>
                    </div>
                  </div>
                )
              })}
            </div>
          </div>
        )}

        {/* 1. QUẢN LÝ TÀI KHOẢN */}
        {currentTab === 'accounts' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4">
              <h3 className="text-base font-extrabold text-gray-900">Cấp tài khoản mới</h3>
              <form onSubmit={handleAddUser} className="grid grid-cols-1 md:grid-cols-5 gap-4">
                <input type="text" placeholder="Mã tài khoản (VD: TT1, PKL)" value={newUsername} onChange={(e) => setNewUsername(e.target.value)} required className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold uppercase" />
                <input type="text" placeholder="Mật khẩu" value={newPassword} onChange={(e) => setNewPassword(e.target.value)} required className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold" />
                <input type="text" placeholder="Họ và tên" value={newFullName} onChange={(e) => setNewFullName(e.target.value)} required className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold" />
                <select value={newRole} onChange={(e) => setNewRole(e.target.value)} className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold cursor-pointer">
                  <option value="to_truong_1">Tổ trưởng 1</option>
                  <option value="to_truong_2">Tổ trưởng 2</option>
                  <option value="to_truong_3">Tổ trưởng 3</option>
                  <option value="to_truong_4">Tổ trưởng 4</option>
                  <option value="pho_ky_luat">Phó kỷ luật</option>
                  <option value="student">Học sinh</option>
                </select>
                <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-2.5 rounded-xl shadow transition text-xs flex items-center justify-center space-x-1 cursor-pointer">
                  <Plus className="h-4 w-4" /><span>Tạo tài khoản</span>
                </button>
              </form>
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th>
                    <th className="p-4">Mã tài khoản</th>
                    <th className="p-4">Họ tên</th>
                    <th className="p-4">Phân quyền</th>
                    <th className="p-4 text-center">Thao tác</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {usersList.map((u, index) => (
                    <tr key={u.id} className="hover:bg-gray-50/50">
                      <td className="p-4">{index + 1}</td>
                      <td className="p-4 font-bold text-purple-700">{u.username}</td>
                      <td className="p-4">{u.full_name}</td>
                      <td className="p-4 uppercase font-bold text-gray-600">{u.role}</td>
                      <td className="p-4 text-center">
                        {u.username !== 'ADMIN' && (
                          <button onClick={() => handleDeleteUser(u.id)} className="bg-red-50 hover:bg-red-100 text-red-600 p-2 rounded-xl transition cursor-pointer">
                            <Trash2 className="h-4 w-4" />
                          </button>
                        )}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* 2. DANH SÁCH HỌC SINH */}
        {currentTab === 'students' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4">
              <h3 className="text-base font-extrabold text-gray-900">Thêm học sinh mới cho lớp {selectedClass}</h3>
              <form onSubmit={handleAddStudent} className="grid grid-cols-1 md:grid-cols-4 gap-4">
                <input type="text" placeholder="Mã học sinh (VD: HS01)" value={newStudentCode} onChange={(e) => setNewStudentCode(e.target.value)} required className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold" />
                <input type="text" placeholder="Họ và tên học sinh" value={newStudentName} onChange={(e) => setNewStudentName(e.target.value)} required className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold" />
                <select value={newStudentTeam} onChange={(e) => setNewStudentTeam(parseInt(e.target.value))} className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold cursor-pointer">
                  <option value={1}>Tổ 1</option>
                  <option value={2}>Tổ 2</option>
                  <option value={3}>Tổ 3</option>
                  <option value={4}>Tổ 4</option>
                </select>
                <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-2.5 rounded-xl shadow transition text-xs flex items-center justify-center space-x-1 cursor-pointer">
                  <Plus className="h-4 w-4" /><span>Thêm học sinh</span>
                </button>
              </form>
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th>
                    <th className="p-4">Mã HS</th>
                    <th className="p-4">Họ và Tên</th>
                    <th className="p-4">Tổ</th>
                    <th className="p-4 text-center">Thao tác</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, index) => (
                    <tr key={st.id} className="hover:bg-gray-50/50">
                      <td className="p-4">{index + 1}</td>
                      <td className="p-4 font-bold text-purple-700">{st.student_code}</td>
                      <td className="p-4 font-bold">{st.full_name}</td>
                      <td className="p-4">Tổ {st.team_number}</td>
                      <td className="p-4 text-center">
                        <button onClick={() => handleDeleteStudent(st.id)} className="bg-red-50 hover:bg-red-100 text-red-600 p-2 rounded-xl transition cursor-pointer">
                          <Trash2 className="h-4 w-4" />
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* 3. THỜI GIAN & TUẦN (HỖ TRỢ NHẬP KÈM NGÀY THÁNG) */}
        {currentTab === 'weeks' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4">
              <h3 className="text-base font-extrabold text-gray-900">Thêm Tuần & Khoảng Thời Gian Đánh Giá</h3>
              <p className="text-xs text-gray-500">Mẹo định dạng chuẩn: <b className="text-purple-600">Tuần 1 - 07/09/2026 - 12/09/2026</b></p>
              
              <form onSubmit={handleAddWeek} className="grid grid-cols-1 md:grid-cols-4 gap-4">
                <input 
                  type="text" 
                  placeholder="VD: Tuần 1 - 07/09/2026 - 12/09/2026" 
                  value={newWeekName} 
                  onChange={(e) => setNewWeekName(e.target.value)} 
                  required 
                  className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold" 
                />
                <select value={newWeekSemester} onChange={(e) => setNewWeekSemester(e.target.value)} className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold cursor-pointer">
                  <option value="HK1">Học Kỳ 1</option>
                  <option value="HK2">Học Kỳ 2</option>
                </select>
                <select value={newWeekPeriod} onChange={(e) => setNewWeekPeriod(e.target.value)} className="px-4 py-2.5 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold cursor-pointer">
                  <option value="GIỮA HỌC KỲ 1">GIỮA HỌC KỲ 1</option>
                  <option value="HỌC KỲ 1">HỌC KỲ 1 (Cuối kỳ)</option>
                  <option value="GIỮA HỌC KỲ 2">GIỮA HỌC KỲ 2</option>
                  <option value="HỌC KỲ 2">HỌC KỲ 2 (Cuối kỳ)</option>
                </select>
                <button type="submit" className="bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-2.5 rounded-xl shadow transition text-xs flex items-center justify-center space-x-1 cursor-pointer">
                  <Plus className="h-4 w-4" /><span>Tạo tuần mới</span>
                </button>
              </form>
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th>
                    <th className="p-4">Tên Tuần & Thời Gian</th>
                    <th className="p-4">Học Kỳ</th>
                    <th className="p-4">Đợt Đánh Giá</th>
                    <th className="p-4 text-center">Thao tác</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {weeksList.map((w, index) => (
                    <tr key={w.id} className="hover:bg-gray-50/50">
                      <td className="p-4">{index + 1}</td>
                      <td className="p-4 font-bold text-purple-700">{w.week_name}</td>
                      <td className="p-4">{w.semester}</td>
                      <td className="p-4 font-bold text-gray-800">{w.period}</td>
                      <td className="p-4 text-center">
                        <button 
                          onClick={async () => {
                            if (confirm(`Thầy có chắc chắn muốn xóa "${w.week_name}" không?`)) {
                              await supabase.from('weeks').delete().eq('id', w.id);
                              fetchAllData();
                            }
                          }}
                          className="bg-red-50 hover:bg-red-100 text-red-600 p-2 rounded-xl transition cursor-pointer"
                          title="Xóa tuần"
                        >
                          <Trash2 className="h-4 w-4" />
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* 4. KẾT QUẢ THI ĐUA TUẦN */}
        {currentTab === 'results' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 flex flex-col md:flex-row justify-between items-center gap-4">
              <div className="flex items-center space-x-3 w-full md:w-auto">
                <span className="text-xs font-extrabold text-gray-700">Chọn Tuần:</span>
                <select
                  value={selectedWeek}
                  onChange={(e) => setSelectedWeek(e.target.value)}
                  className="px-4 py-2.5 bg-purple-50 border border-purple-200 rounded-xl font-bold text-purple-900 text-xs cursor-pointer outline-none"
                >
                  {weeksList.map((w) => (
                    <option key={w.id} value={w.id}>{w.week_name} ({w.period})</option>
                  ))}
                </select>
              </div>
              <button
                onClick={exportToExcel}
                className="w-full md:w-auto bg-green-600 hover:bg-green-700 text-white font-extrabold px-6 py-2.5 rounded-xl shadow transition text-xs flex items-center justify-center space-x-2 cursor-pointer"
              >
                <FileSpreadsheet className="h-4 w-4" />
                <span>Xuất Báo Cáo Excel</span>
              </button>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
              {[1, 2, 3, 4].map((tNum) => (
                <div key={tNum} className="bg-white p-4 rounded-2xl border border-purple-100 shadow-sm flex flex-col items-center justify-between space-y-3">
                  <h4 className="text-xs font-black text-purple-900 uppercase">Ảnh vi phạm Tổ {tNum}</h4>
                  {teamImages[tNum] ? (
                    <div className="w-full flex flex-col space-y-2">
                      <button onClick={() => { setZoom(1); setRotation(0); setPreviewImage(teamImages[tNum]); }} className="w-full bg-purple-50 hover:bg-purple-100 text-purple-700 font-bold py-2 rounded-xl text-xs flex items-center justify-center space-x-1 transition cursor-pointer">
                        <ImageIcon className="h-4 w-4" /><span>Xem ảnh Tổ {tNum}</span>
                      </button>
                      <button onClick={() => handleDeleteTeamImage(tNum)} className="w-full bg-red-50 hover:bg-red-100 text-red-600 font-bold py-1.5 rounded-xl text-xs flex items-center justify-center space-x-1 transition cursor-pointer">
                        <Trash2 className="h-3.5 w-3.5" /><span>Xóa/Thu hồi ảnh</span>
                      </button>
                    </div>
                  ) : (
                    <span className="text-xs text-gray-400 italic py-2">Chưa có ảnh</span>
                  )}
                </div>
              ))}
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th>
                    <th className="p-4">Họ và Tên</th>
                    <th className="p-4">Tổ</th>
                    <th className="p-4">Tổng Điểm</th>
                    <th className="p-4">Xếp Loại</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, index) => {
                    const sc = scoresList.find(s => s.student_id === st.id) || { score: 100, evaluation: 'T' }
                    return (
                      <tr key={st.id} className="hover:bg-gray-50/50">
                        <td className="p-4">{index + 1}</td>
                        <td className="p-4 font-bold text-gray-900">{st.full_name}</td>
                        <td className="p-4">Tổ {st.team_number}</td>
                        <td className="p-4">
                          <input
                            type="number"
                            value={sc.score}
                            onChange={(e) => handleScoreUpdate(st.id, parseFloat(e.target.value) || 0)}
                            className="w-24 px-3 py-1.5 bg-gray-50 border border-gray-200 rounded-lg font-bold text-center text-purple-700 outline-none"
                          />
                        </td>
                        <td className="p-4 font-extrabold">
                          <span className={`px-2.5 py-1 rounded-lg text-xs ${
                            sc.evaluation === 'T' ? 'bg-green-100 text-green-700' :
                            sc.evaluation === 'K' ? 'bg-blue-100 text-blue-700' :
                            sc.evaluation === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'
                          }`}>
                            {sc.evaluation}
                          </span>
                        </td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* 5. THỐNG KÊ THEO ĐỢT */}
        {currentTab === 'periods' && (
          <div className="space-y-6">
            <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 flex items-center space-x-4">
              <span className="text-xs font-extrabold text-gray-700">Chọn Đợt Thống Kê:</span>
              <select
                value={selectedPeriod}
                onChange={(e) => setSelectedPeriod(e.target.value)}
                className="px-4 py-2.5 bg-purple-50 border border-purple-200 rounded-xl font-bold text-purple-900 text-xs cursor-pointer outline-none"
              >
                <option value="GIỮA HỌC KỲ 1">GIỮA HỌC KỲ 1</option>
                <option value="HỌC KỲ 1">HỌC KỲ 1</option>
                <option value="GIỮA HỌC KỲ 2">GIỮA HỌC KỲ 2</option>
                <option value="HỌC KỲ 2">HỌC KỲ 2</option>
                <option value="CẢ NĂM">CẢ NĂM</option>
              </select>
            </div>

            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 overflow-hidden">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-purple-50 text-purple-900 text-xs font-extrabold">
                    <th className="p-4">STT</th>
                    <th className="p-4">Họ và Tên Học Sinh</th>
                    <th className="p-4">Tổ</th>
                    <th className="p-4">Điểm Trung Bình Đợt</th>
                    <th className="p-4">Xếp Loại Đợt</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 text-xs font-medium text-gray-700">
                  {studentsList.map((st, index) => {
                    const filteredWeeks = weeksList.filter(w => {
                      if (selectedPeriod === 'GIỮA HỌC KỲ 1') return w.period === 'GIỮA HỌC KỲ 1';
                      if (selectedPeriod === 'HỌC KỲ 1') return w.semester === 'HK1';
                      if (selectedPeriod === 'GIỮA HỌC KỲ 2') return w.period === 'GIỮA HỌC KỲ 2';
                      if (selectedPeriod === 'HỌC KỲ 2') return w.semester === 'HK2';
                      if (selectedPeriod === 'CẢ NĂM') return true;
                      return false;
                    });

                    const weekIds = filteredWeeks.map(w => w.id)
                    const studentScoresInPeriod = scoresList.filter(s => s.student_id === st.id && weekIds.includes(s.week_id))
                    
                    const avgScore = studentScoresInPeriod.length > 0 
                      ? Math.round(studentScoresInPeriod.reduce((acc, curr) => acc + curr.score, 0) / studentScoresInPeriod.length) 
                      : 100
                    const evalPeriod = getEvaluation(avgScore)

                    return (
                      <tr key={st.id} className="hover:bg-gray-50/50">
                        <td className="p-4">{index + 1}</td>
                        <td className="p-4 font-bold text-gray-900">{st.full_name}</td>
                        <td className="p-4">Tổ {st.team_number}</td>
                        <td className="p-4 font-bold text-purple-700">{avgScore} điểm</td>
                        <td className="p-4 font-extrabold">
                          <span className={`px-2.5 py-1 rounded-lg text-xs ${
                            evalPeriod === 'T' ? 'bg-green-100 text-green-700' :
                            evalPeriod === 'K' ? 'bg-blue-100 text-blue-700' :
                            evalPeriod === 'TB' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'
                          }`}>
                            {evalPeriod}
                          </span>
                        </td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* 6. TRA CỨU SAO ĐỎ */}
        {currentTab === 'saodo' && (
          <div className="bg-white p-8 rounded-3xl shadow-sm border border-gray-100 text-center space-y-4">
            <h3 className="text-lg font-black text-gray-900">Hệ Thống Tra Cứu Vi Phạm Sao Đỏ</h3>
            <p className="text-xs text-gray-500">Truy cập trực tiếp trang tra cứu điểm thi đua chính thức của trường.</p>
            <div>
              <a
                href="https://thiduanamhoc2627.netlify.app/"
                target="_blank"
                rel="noopener noreferrer"
                className="inline-flex items-center space-x-2 bg-purple-600 hover:bg-purple-700 text-white font-extrabold px-6 py-3 rounded-xl shadow transition text-xs"
              >
                <span>Mở trang tra cứu Sao Đỏ</span>
                <ExternalLink className="h-4 w-4" />
              </a>
            </div>
          </div>
        )}

        {/* 7. SỔ KỶ LUẬT GIÁO VIÊN BỘ MÔN */}
        {currentTab === 'teachers' && (
          <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4">
            <h3 className="text-base font-extrabold text-gray-900">Sổ Kỷ Luật Ghi Nhận Từ Giáo Viên Bộ Môn</h3>
            <div className="space-y-3">
              {disciplinaryList.map((d) => (
                <div key={d.id} className="p-4 bg-gray-50 rounded-2xl border border-gray-100 space-y-2 flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
                  <div className="space-y-1">
                    <p className="text-xs font-bold text-gray-900">{d.content}</p>
                    <p className="text-[11px] text-gray-500 font-medium">Giáo viên: <b>{d.teacher_name || 'Không rõ'}</b> | Người đăng: {d.creator_name || 'PKL'}</p>
                  </div>
                  <div className="flex items-center space-x-3 w-full md:w-auto justify-end">
                    {d.image_url && (
                      <button onClick={() => { setZoom(1); setRotation(0); setPreviewImage(d.image_url); }} className="text-purple-600 font-bold flex items-center space-x-1 text-xs bg-purple-50 px-3 py-1.5 rounded-xl hover:bg-purple-100 transition cursor-pointer">
                        <ImageIcon className="h-3.5 w-3.5" /><span>Xem ảnh</span>
                      </button>
                    )}
                    <button onClick={() => handleDeleteDisciplinary(d.id)} className="bg-red-50 hover:bg-red-100 text-red-600 p-2 rounded-xl transition cursor-pointer" title="Xóa kỷ luật">
                      <Trash2 className="h-4 w-4" />
                    </button>
                  </div>
                </div>
              ))}
            </div>
          </div>
        )}

        {/* 8. CÀI ĐẶT LÀM MỚI DỮ LIỆU */}
        {currentTab === 'settings' && (
          <div className="bg-white p-6 rounded-3xl shadow-sm border border-gray-100 space-y-4 max-w-xl">
            <h3 className="text-base font-extrabold text-red-600">Làm Mới Dữ Liệu (Sang Năm Học Mới)</h3>
            <p className="text-xs text-gray-500">Thao tác này sẽ xóa sạch toàn bộ điểm số và ảnh vi phạm cũ. Yêu cầu nhập mật khẩu Admin để xác nhận.</p>
            <div className="space-y-3">
              <input
                type="password"
                placeholder="Nhập mật khẩu Admin (luuvanglien0410)"
                value={resetPasswordInput}
                onChange={(e) => setResetPasswordInput(e.target.value)}
                className="w-full px-4 py-3 bg-gray-50 border border-gray-200 rounded-xl text-xs font-semibold outline-none"
              />
              <button
                onClick={handleResetSystem}
                className="w-full bg-red-600 hover:bg-red-700 text-white font-extrabold py-3 rounded-xl shadow transition text-xs cursor-pointer"
              >
                Xác Nhận Xóa Dữ Liệu Cũ
              </button>
            </div>
          </div>
        )}

      </main>

      {/* Modal Xem Ảnh Trực Tiếp (Phóng to, Thu nhỏ, Xoay ảnh) */}
      {previewImage && (
        <div className="fixed inset-0 bg-black/80 z-50 flex flex-col items-center justify-center p-4">
          <div className="absolute top-4 right-4 flex space-x-3">
            <button onClick={() => setZoom(z => z + 0.2)} className="bg-white/20 hover:bg-white/40 text-white px-3 py-1.5 rounded-xl font-bold text-xs cursor-pointer">Phóng to (+)</button>
            <button onClick={() => setZoom(z => Math.max(0.4, z - 0.2))} className="bg-white/20 hover:bg-white/40 text-white px-3 py-1.5 rounded-xl font-bold text-xs cursor-pointer">Thu nhỏ (-)</button>
            <button onClick={() => setRotation(r => (r + 90) % 360)} className="bg-white/20 hover:bg-white/40 text-white px-3 py-1.5 rounded-xl font-bold text-xs cursor-pointer">Xoay ảnh</button>
            <button onClick={() => setPreviewImage(null)} className="bg-red-600 hover:bg-red-700 text-white px-4 py-1.5 rounded-xl font-bold text-xs cursor-pointer">Đóng (X)</button>
          </div>
          <div className="overflow-auto max-w-full max-h-[85vh] flex items-center justify-center p-2">
            <img src={previewImage} alt="Xem chi tiết" style={{ transform: `scale(${zoom}) rotate(${rotation}deg)`, transition: 'transform 0.2s ease' }} className="max-h-[80vh] object-contain rounded-lg shadow-2xl" />
          </div>
        </div>
      )}
    </div>
  )
}
